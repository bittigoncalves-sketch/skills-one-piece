# ============================================================================
#  ESTILO DE COMBATE BASE — Z e X feitas com o Skills Builder (14/10/2026). Fonte reproduzível.
#
#    "$(./find_godot.sh)" --headless --path . --script tools/criar_estilo_base.gd [-- <pasta>]
#
#  Pedido do dono: o antigo estilo "corpo a corpo" (item "Luta" da barra) vira o Estilo de
#  Combate Base. Z = soco poderoso que lança uma rajada de vento com knockback. X = soco no
#  chão que abre uma rachadura e explode em vento em volta do jogador, empurrando os inimigos.
#
#  Mesmo caminho da Nikyu (`criar_nikyu_nikyu.gd`): Dados.novo + blocos + poses, publicado
#  pela API do editor. O catálogo guarda a entrada com `estilo: true` (não é fruta à venda); o
#  item "Luta" a equipa (`ProgressaoDoPersonagem.fruta_a_equipar`). Rodar de novo republica.
#  Modelos: `tools/blender/rajada_vento.py` e `rachadura_chao.py`.
#  Números em escala da ARENA (a história multiplica por ESCALA_DA_ARENA = 250/2048).
# ============================================================================

const Dados := preload("res://src/skills_builder/dados_skill.gd")
const NOME := "Estilo de Combate Base"
const COR := "#bfe6ff"

var _falhas := 0


func _init() -> void:
	_ir.call_deferred()


func _ir() -> void:
	await process_frame
	var args := OS.get_cmdline_user_args()
	if args.size() > 0:
		SkillsPublicadas.pasta_forcada = args[0]
		SkillsPublicadas.recarregar()
	var fid := SkillsPublicadas.id_de_nome(NOME)
	if fid != KitDoBuilder.ID_ESTILO_BASE:
		push_error("id do estilo mudou: %s" % fid)
		quit(1)
		return
	if SkillsPublicadas.fruta(fid).is_empty():
		var r := SkillsPublicadas.criar_fruta(NOME, COR, true)
		if not r.ok:
			push_error("criar estilo: %s" % str(r.erros))
			quit(1)
			return
	for s in [_z(fid), _x(fid)]:
		var p := SkillsPublicadas.publicar(s)
		print("%s %s %s" % ["OK  " if p.ok else "ERRO", s.identificacao.id, str(p.get("erros", []))])
		if not p.ok:
			_falhas += 1
	quit(1 if _falhas > 0 else 0)


func _base(fid: String, id: String, nome: String, desc: String, tecla: String) -> Dictionary:
	var s := Dados.novo()
	s.eventos = []
	s.identificacao.merge({"id": id, "nome": nome, "descricao": desc, "dono": "Jogador", "fruta": fid}, true)
	# Segurar a tecla trava o corpo ONDE ESTÁ, gravidade incluída (regra do dono, 14/10/2026).
	s.ativacao.merge({"modo": "Pressionar", "tecla": tecla, "paralisar_segurando": true}, true)
	s.condicoes.merge({"exige_alvo": false, "distancia_maxima": 60.0}, true)
	s.vfx.merge({"cor": COR, "cor_2": "#ffffff"}, true)
	s.animacao.merge({"clipe": "Ataque", "pose": ""}, true)
	return s


func _pose(id: String, nome: String, ossos: Dictionary) -> Dictionary:
	return {"id": id, "nome": nome, "ossos": ossos}


func _ev(tipo: String, id: String, inicio: float, duracao: float, parametros: Dictionary) -> Dictionary:
	return {"id": id, "tipo": tipo, "inicio": inicio, "duracao": duracao, "habilitado": true, "parametros": parametros}


func _pose_ev(id: String, inicio: float, duracao: float, pose: String, transicao := 0.1) -> Dictionary:
	return _ev("animacao", id, inicio, duracao, {"pose": pose, "transicao": transicao})


func _som(id: String, inicio: float, som: String, tom: float, volume := -6.0) -> Dictionary:
	return _ev("audio", id, inicio, 0.3, {"som": som, "tom": tom, "volume": volume})
